// [구글 Apps Script 전용 코드 - 시트 DB 연동 & 개인정보 보호] function doPost(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var data = JSON.parse(e.postData.contents); var action = data.action; // 'login' 또는 'save' var name = String(data.name || "").trim(); var studentId = String(data.studentId || "").trim(); var score = Number(data.score || 0); var passed = score >= 90 ? "통과" : "미통과"; var rows = sheet.getDataRange().getValues(); var foundRowIndex = -1; // 2행부터 기존 학생 데이터 검색 (이름 + 학번 일치 여부 확인) for (var i = 1; i < rows.length; i++) { if (String(rows[i][0]).trim() === name && String(rows[i][1]).trim() === studentId) { foundRowIndex = i + 1; // 시트 행 번호는 1부터 시작 break; } } // [기능 1] 접속 시 신원 확인 (웹앱에는 이름/학번을 돌려주지 않고 점수/통과여부만 반환!) if (action === "login") { if (foundRowIndex !== -1) { var savedScore = Number(rows[foundRowIndex - 1][2] || 0); var savedPassed = String(rows[foundRowIndex - 1][3] || "미통과"); return ContentService.createTextOutput(JSON.stringify({ status: "existing", score: savedScore, passed: savedPassed === "통과" })).setMimeType(ContentService.MimeType.JSON); } else { // 신규 학생: 시트에 초기 행 생성 (이름, 학번, 0, 미통과) sheet.appendRow([name, studentId, 0, "미통과"]); return ContentService.createTextOutput(JSON.stringify({ status: "new", score: 0, passed: false })).setMimeType(ContentService.MimeType.JSON); } } // [기능 2] 퀴즈 점수 업데이트 저장 (최고 점수 유지) if (action === "save") { if (foundRowIndex !== -1) { var prevScore = Number(rows[foundRowIndex - 1][2] || 0); var bestScore = Math.max(prevScore, score); var bestPassed = bestScore >= 90 ? "통과" : "미통과"; sheet.getRange(foundRowIndex, 3).setValue(bestScore); sheet.getRange(foundRowIndex, 4).setValue(bestPassed); } else { sheet.appendRow([name, studentId, score, passed]); } return ContentService.createTextOutput(JSON.stringify({ status: "saved" })).setMimeType(ContentService.MimeType.JSON); } }
